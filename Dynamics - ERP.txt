ANALISIS DE LIQUIDEZ
--------------------

Liquidez General = [Activo Circulante]/[Pasivo Circulante]
Prueba Acida = [Activo Circulante (- inv)]/[Pasivo Circulante]


Rotacion de cuentas por cobrar = CALCULATE([Total Activos],FILTER(data,data[NIVEL 4]="SERVICIOS POR COBRAR"))/
						(CALCULATE([Total Ingresos],FILTER(data,data[NIVEL 2]="INGRESOS OPERATIVOS"))/360)

Rotacion de Inventario = CALCULATE([Total Activos],FILTER(data,data[NIVEL 3]="BIENES DE CAMBIO O REALIZABLE"))/([Total Costos]/360)


ANALISIS DE ENDEUDAMIENTO
-------------------------
Endeudamiento Patrimonial = [Total Pasivos]/[Total Patrimonio]
Endeudamiento a Corto Plazo = CALCULATE([Total Pasivos],FILTER(data,data[NIVEL 2]="PASIVO CORRIENTE"))/[Total Patrimonio]
Endeudamiento a Largo Plazo = CALCULATE([Total Pasivos],FILTER(data,data[NIVEL 2]="PASIVO NO CORRIENTE"))/[Total Patrimonio]
Endeudamiento del Activo = [Total Pasivos]/[Total Activos]

RENTABILIDAD

QUERY PARA VENTAS

SELECT     
    T1.DocEntry AS 'ID',    
    T1.DocNum AS 'Folio', 
    T1.ObjType AS 'Tipo Documento', 
    CASE when T1.CANCELED = 'Y' then 'Cancelled' ELse 'Not Cancelled' END AS [Cancelada],
    CASE when T1.DocStatus = 'C' then 'Closed' Else 'Open' end AS [DocStatus],
    CASE T1.ObjType 
        WHEN 13 THEN 'Factura' 
        WHEN 14 THEN 'NC' 
        END AS 'Tipo Documento', 
    T1.Confirmed AS 'Corfirmada',
    T1.TransId AS 'TransaccionID', 
    T1.DocDate AS 'Fecha Creacion',
    T0.ShipDate AS 'Fecha Detalle', 
    T0.ItemCode AS 'productoID', 
    T0.Dscription AS 'Producto', 
    T0.Quantity AS 'Cantidad', 
    T0.Price AS 'Precio', 
    T0.Currency AS 'Moneda', 
    T0.LineTotal AS 'Monto', 
    T0.SlpCode AS 'VendedorID', 
    T2.SlpName AS 'Vendedor', 
    T4.ItmsGrpCod AS 'CategoriaID', 
    T4.ItmsGrpNam AS 'Categoria', 
    T5.FirmCode AS 'ProveedorID', 
    T5.FirmName AS 'Proveedor', 
    T1.CardCode AS 'clienteID', 
    T1.CardName AS 'Cliente', 
    T1.Address AS 'Direccion Cliente'
FROM        
    INV1 T0 INNER JOIN
    OINV T1 ON T0.DocEntry = T1.DocEntry INNER JOIN
    OSLP T2 ON T1.SlpCode = T2.SlpCode INNER JOIN
    OITM T3 ON T0.ItemCode = T3.ItemCode INNER JOIN
    OITB T4 ON T3.ItmsGrpCod = T4.ItmsGrpCod INNER JOIN
    OMRC T5 ON T3.FirmCode = T5.FirmCode



Query SQL tabla Compras

SELECT
T0.[DocNum] AS 'Pedido',
T4.[Docnum] AS 'Factura', 
T0.[DocDate] AS 'FechaDoc',
T0.[DocDueDate] AS 'Fecha Vencimiento', 
CASE when T0.CANCELED = 'Y' then 'Cancelada' else 'No cancelada' END as [Cancelada],
T0.[SlpCode] AS 'AutorID', 
T5.SlpName AS 'Analista',
T1.[ItemCode] AS 'ItemID', 
T1.[Dscription] AS 'Producto', 
T7.ItmsGrpCod AS 'CategoriaID', 
T7.ItmsGrpNam AS 'Categoria',
T1.[Quantity] AS 'Cantidad pedida', 
T1.Price AS 'Precio unitario',
T1.LineTotal AS 'Monto total',
T1.[OpenQty] AS 'Cantidad por recibir',
T3.[DocNum] AS 'Doc Entrada', 
T3.[DocDate] AS 'Fecha Entrada', 
CASE T1.LineStatus
    WHEN 'C' THEN 'Cerrado'
    WHEN 'O' THEN 'Abierto'
    END AS 'Estado de linea',
T0.[CardCode] AS 'Cod Proveedor',
T0.[CardName] AS 'Nombre Proveedor',
T0.[Address] AS 'Direccion Proveedor',
T4.[PeyMethod] AS 'Via de pago',
T1.Currency AS 'Moneda'
FROM OPOR T0
INNER JOIN POR1 T1 ON T0.DocEntry = T1.DocEntry
left outer join PDN1 T2 on T2.BaseEntry=T1.Docentry and T2.BaseLine = T1.LineNum
Left outer JOIN OPDN T3 ON T2.DocEntry = T3.DocEntry
left join OPCH T4 on T0.docentry=T4.docentry
left join OSLP T5 ON T0.SlpCode = T5.SlpCode 
left join OITM T6 ON T1.ItemCode = T6.ItemCode 
left join OITB T7 ON T6.ItmsGrpCod = T7.ItmsGrpCod

SELECT 
T0.CardCode AS 'ClienteID',
T0.CntctPrsn AS 'Nombre del Cliente',
T0.CardName AS 'Razon Social',
T0.Phone1 AS 'Telefono',
T0.Address AS 'Direccion',
T0.City AS 'Ciudad',
T0.Country AS 'Pais',
T0.Currency AS 'Moneda',
T0.E_Mail AS 'email',
T0.CreateDate AS 'Fecha de Registro',
T0.ShipToDef AS 'Tipo de entrega',
T0.Deleted AS 'Eliminado',
T0.Priority AS 'Prioridad',
T0.CreditLine AS 'Linea de credito',
CASE T0.CmpPrivate 
        WHEN 'C' THEN 'Compañia' 
        WHEN 'G' THEN 'Gobierno'
        WHEN 'I' THEN 'Particular'
        END AS 'Tipo Empresa',
CASE T0.CardType
        WHEN 'C' THEN 'Cliente'
        WHEN 'L' THEN 'Lead'
        WHEN 'S' THEN 'Vendedor'
        END AS 'Tipo Cliente'
FROM 
    OCRD T0
