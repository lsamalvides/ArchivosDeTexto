VAR MaxFecha = CALCULATE(MAX(CALENDARIO[DATE]),
	ALL(CALENDARIO))
VAR MinFecha = DATE(YEAR(MaxFecha), Month(MaxFecha), -2, 1)
VAR Resultado =
	CALCULATE(
		[VENTAS],
		FILTER(
			CALENDARIO,
			CALENDARIO[DATE] >= MinFecha && CALENDARIO[DATE] <= MaxFecha
		)
)
Return
Resultado
